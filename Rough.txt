from __future__ import annotations

import concurrent.futures as cf
from pathlib import Path

from ..util import log
from . import defi_rules, scan, solidity, tools

SEVERITY_ORDER = {"critical": 4, "high": 3, "medium": 2, "low": 1, "info": 0}
BASE_CVSS = {"critical": 9.4, "high": 7.8, "medium": 5.4, "low": 2.9, "info": 0.0}
CONF_ADJ = {"high": 0.4, "medium": 0.0, "low": -0.6}


def cvss_for(severity: str, confidence: str) -> float:
    return round(max(0.0, min(10.0, BASE_CVSS.get(severity, 5.0) + CONF_ADJ.get(confidence, 0.0))), 1)


def economic_multiplier(f: dict) -> float:
    base = {"reentrancy": 1.35, "oracle": 1.4, "flashloan": 1.35, "accounting": 1.3,
            "precision": 1.2, "signature": 1.3, "upgrade": 1.25,
            "access-control": 1.3}.get(f.get("category", ""), 1.0)
    return base * (1 + 0.1 * (int(f.get("economic", 1)) - 1))


def normalise(f: dict) -> dict:
    sev = str(f.get("severity") or "medium").lower().strip()
    sev = {"moderate": "medium", "error": "high", "warning": "medium", "note": "low"}.get(sev, sev)
    f["severity"] = sev if sev in SEVERITY_ORDER else "medium"
    conf = str(f.get("confidence") or "medium").lower()
    f["confidence"] = conf if conf in CONF_ADJ else "medium"
    f["cvss"] = round(min(10.0, cvss_for(f["severity"], f["confidence"]) * economic_multiplier(f)), 1)
    return f


def dedupe(findings: list) -> list:
    """Collapse to one hit per rule per file, keeping the highest-scoring line.
    Line numbers shift with refactors; the rule+file pair is the stable identity."""
    best = {}
    for f in findings:
        key = f"{f.get('file')}::{f.get('tool')}::{(f.get('title') or '')[:40].lower()}"
        if key not in best or f["cvss"] > best[key]["cvss"]:
            best[key] = f
    return list(best.values())


def cap_per_rule(findings: list, cap: int = 3, max_total: int = 120) -> list:
    """One rule must not own the report, and the report must not run to hundreds."""
    counts: dict = {}
    out = []
    for f in findings:
        key = f.get("tool") or "?"
        counts[key] = counts.get(key, 0) + 1
        if counts[key] <= cap:
            out.append(f)
        if len(out) >= max_total:
            break
    return out


def _norm_path(p: str) -> str:
    return (p or "").replace("\\", "/").lstrip("./").strip("/")


def in_scope(f: dict, scope_paths) -> bool:
    """Findings carry repo-relative paths; build findings carry none."""
    if not scope_paths:
        return True
    path = _norm_path(f.get("file") or "")
    if not path:
        return True
    for p in scope_paths:
        p = _norm_path(p)
        if p and (path == p or path.startswith(p + "/")):
            return True
    return False


def run_audit(cfg, repo: Path, rules_dir: Path, solidity_mode: bool = False,
              scope_paths=None) -> dict:
    enabled = {k for k, v in (cfg.audit.get("tools") or {}).items() if v}
    sol_enabled = {k for k, v in (cfg.solidity.get("tools") or {}).items() if v} if solidity_mode else set()
    all_findings, tool_report = [], []

    with cf.ThreadPoolExecutor(max_workers=max(1, cfg.max_concurrency)) as pool:
        futures = {}
        for name in enabled:
            if name not in tools.RUNNERS:
                continue
            if name == "semgrep":
                futures[pool.submit(tools.RUNNERS[name], repo, cfg, rules_dir)] = name
            else:
                futures[pool.submit(tools.RUNNERS[name], repo, cfg)] = name
        for name in sol_enabled:
            if name in solidity.RUNNERS:
                futures[pool.submit(solidity.RUNNERS[name], repo, cfg)] = name
        for fut in cf.as_completed(futures):
            name = futures[fut]
            try:
                found, meta = fut.result()
            except Exception as exc:
                tool_report.append({"tool": name, "status": f"crash: {exc}"})
                continue
            all_findings.extend(found)
            tool_report.append(meta)
            log.info("%s -> %d (%s)", name, len(found), meta.get("status"))

    if cfg.audit.get("custom_rules", True):
        hits = scan.run_rules(repo, cfg)
        all_findings.extend(hits)
        tool_report.append({"tool": "custom-rules", "status": "ok", "count": len(hits)})
        log.info("custom rules -> %d", len(hits))
    if solidity_mode:
        hits = defi_rules.run(repo, cfg)
        all_findings.extend(hits)
        tool_report.append({"tool": "defi-rules", "status": "ok", "count": len(hits)})
        log.info("defi rules -> %d", len(hits))

    floor = SEVERITY_ORDER.get(str(cfg.audit.get("severity_floor", "low")).lower(), 1)
    norm = [normalise(f) for f in all_findings]
    norm = dedupe([f for f in norm if SEVERITY_ORDER[f["severity"]] >= floor])
    norm = [f for f in norm if in_scope(f, scope_paths)]
    norm.sort(key=lambda f: (-f["cvss"], f.get("file") or "", f.get("line") or 0))
    norm = cap_per_rule(norm,
                        int(cfg.audit.get("max_hits_per_rule", 3)),
                        int(cfg.audit.get("max_total_findings", 120)))

    counts = {s: 0 for s in SEVERITY_ORDER}
    for f in norm:
        counts[f["severity"]] += 1
    for i, f in enumerate(norm, 1):
        f["fid"] = f"{f['severity'][:4].upper()}-{i:03d}"
    return {"findings": norm, "counts": counts, "tools": tool_report}

PRAGMA_RX = re.compile(r"pragma\s+solidity\s+[^0-9]*([0-9]+\.[0-9]+\.[0-9]+)")


def _pragma(repo: Path | None) -> str:
    """Match the target's pragma - a hardcoded 0.8.20 will not compile a 0.7.x tree."""
    if repo is not None and repo.exists():
        seen = []
        for sol in list(repo.rglob("*.sol"))[:400]:
            if any(p in {"test", "lib", "node_modules"} for p in sol.parts):
                continue
            try:
                m = PRAGMA_RX.search(sol.read_text(encoding="utf-8", errors="replace")[:4000])
            except OSError:
                continue
            if m:
                seen.append(m.group(1))
        if seen:
            return max(seen, key=lambda v: tuple(int(x) for x in v.split(".")))
    return "0.8.20"


def _name(fid: str, title: str) -> str:
    stem = re.sub(r"[^A-Za-z0-9]+", "_", title)[:40].strip("_")
    return f"PoC_{fid.replace('-', '')}_{stem}"


def generate(project_path: Path, findings: list, scope: dict,
             fallback_address: str | None = None, repo: Path | None = None) -> list:
    out_dir = project_path / "poc" / "test"
    out_dir.mkdir(parents=True, exist_ok=True)
    pragma = _pragma(repo)
    addrs = scope.get("addresses") or ([fallback_address] if fallback_address else [])
    written, used = [], set()
    for f in findings:
        if len(written) >= 10:
            break
        if f["severity"] not in ("critical", "high"):
            continue
        name = _name(f["fid"], f["title"])
        while name in used:
            name += "_"
        used.add(name)
        body = TEMPLATE.format(pragma=pragma, title=f["title"], fid=f["fid"],
                               severity=f["severity"], cvss=f["cvss"],
                               file=f.get("file") or "?",
                               contract=name, target=addrs[0] if addrs else "address(0)")
        p = out_dir / f"{name}.t.sol"
        p.write_text(body, encoding="utf-8")
        written.append(p)
    return written

def cmd_target(args):
    cfg, workspace, db, session = _ctx(args)
    console.print(f"[cyan]intake[/] {args.url}")
    pid, scope = intake(cfg, db, session, args.url)

    t = Table(title="parsed scope")
    t.add_column("field")
    t.add_column("value", overflow="fold")
    for r in scope["repos"]:
        t.add_row("repo", r["url"])
        for p_ in r["paths"]:
            t.add_row("", f"  path: {p_}")
        if not r["paths"]:
            t.add_row("", "  (whole repository)")
    t.add_row("ladder", "\n".join(f"{s['severity']}: ${s['max_reward']:,}"
                                  for s in scope["severity_table"]) or "-")
    t.add_row("contacts", ", ".join(scope["contacts"]) or "-")
    t.add_row("exclusions", str(len(scope["exclusions"])))
    console.print(t)

    p = db.get_project(str(pid))
    pdir = project_dir(workspace, p["platform"], p["slug"])
    console.print(f"[green]scope ->[/] {save_scope(pdir, scope)}")
    if not args.go:
        console.print("\n[dim]re-run with --go to clone and audit[/]")
        return

    if not os.environ.get(cfg.solidity.get("fork_rpc_env", "ETH_RPC_URL")):
        console.print("[yellow]ETH_RPC_URL unset - PoC scaffolds will not fork[/]")

    root = pdir / "repo"
    findings, cloned, failed = [], [], []
    repo_paths: dict = {}
    for r in scope["repos"]:
        name = r["url"].rstrip("/").split("/")[-1].removesuffix(".git")
        dest = root / name
        try:
            clone_repo(r["url"], r["branch"], dest)
            cloned.append(dest)
            repo_paths[dest] = r["paths"]
        except CloneError as exc:
            console.print(f"[red]{name}: {exc}[/]")
            failed.append(name)

    prior: list = []
    for dest in cloned:
        prior += find_prior_audits(dest)
    if prior:
        scope["existing_audits"] = prior
        console.print(f"[yellow]{len(prior)} prior audit file(s) in repo - read these first[/]")
        for a in prior[:8]:
            console.print(f"  [dim]{a}[/]")

    all_tools: list = []
    for dest in cloned:
        stats = repo_stats(dest)
        console.print(f"[cyan]auditing[/] {dest.name} ({stats['files']} files)")
        res = run_audit(cfg, dest, RULES_DIR, solidity_mode=True,
                        scope_paths=repo_paths.get(dest) or None)
        console.print(_tool_table(dest.name, res["tools"]))
        dead = _dead_tools(res["tools"])
        if dead:
            console.print(f"[red]{dest.name}: {', '.join(dead)} failed - tree not compiled[/]")
            failed.append(dest.name)
            all_tools.extend(res["tools"])
            continue
        for f in res["findings"]:
            f["file"] = f"{dest.name}/{f.get('file') or ''}"
        findings.extend(res["findings"])
        all_tools.extend(res["tools"])
        for t_ in res["tools"]:
            db.add_run(pid, t_.get("tool", "?"), t_.get("status", "?"))

    if not findings:
        console.print(f"\n[red]no findings - {len(failed)} of {len(cloned) + len(failed)} "
                      f"repos failed[/]")
        db.set_status(pid, "error")
        sys.exit(1)

    findings.sort(key=lambda x: -x["cvss"])
    for f in findings:
        db.add_finding(pid, f)

    counts = {s: 0 for s in ("critical", "high", "medium", "low", "info")}
    for f in findings:
        counts[f["severity"]] += 1
    audit = {"findings": findings, "counts": counts, "tools": all_tools}

    for path in write_reports(p, audit, cfg, pdir, {"files": 0, "bytes": 0}, scope):
        console.print(f"[green]report ->[/] {path}")
    if cfg.poc.get("enabled", True):
        fallback = None
        for dest in cloned:
            fallback = first_deployment_address(dest)
            if fallback:
                break
        for path in generate(pdir, findings, scope, fallback,
                             repo=cloned[0] if cloned else None):
            console.print(f"[green]poc ->[/] {path}")
    db.set_status(pid, "audited")

    if counts["critical"] or counts["high"]:
        console.print(f"\n[yellow]critical {counts['critical']} / high {counts['high']} "
                      f"- validate against the live target before writing up[/]")

"""Repository acquisition.

Full blobless clone - no sparse checkout - so cross-directory Solidity
imports resolve. Re-running over an existing checkout updates in place
instead of dying on 'destination path already exists'.
"""
from __future__ import annotations

import os
import shutil
import subprocess
import time
from datetime import datetime, timezone
from pathlib import Path

RETRIES = 3


class CloneError(RuntimeError):
    """Raised when a repository cannot be acquired."""


def _run(argv, cwd=None):
    """Resolve the executable and wrap .cmd/.bat - CreateProcess cannot exec them."""
    exe = shutil.which(argv[0])
    if exe:
        argv = [exe, *argv[1:]]
        if os.name == "nt" and exe.lower().endswith((".cmd", ".bat")):
            argv = ["cmd", "/c", *argv]
    return subprocess.run(
        argv,
        cwd=cwd,
        capture_output=True,
        text=True,
        encoding="utf-8",
        errors="replace",
    )


def _git(*args, cwd=None):
    return _run(["git", *args], cwd=cwd)


def _quarantine(dest: Path) -> None:
    stamp = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ")
    dest.rename(dest.with_name(f"{dest.name}.bak-{stamp}"))


def _default_ref(dest: Path):
    ref = _git(
        "-C", str(dest), "symbolic-ref", "--quiet", "refs/remotes/origin/HEAD"
    ).stdout.strip()
    if ref:
        return ref.replace("refs/remotes/", "")
    for cand in ("origin/main", "origin/master"):
        if _git("-C", str(dest), "rev-parse", "--verify", "--quiet", cand).returncode == 0:
            return cand
    out = _git(
        "-C", str(dest), "for-each-ref", "--format=%(refname:short)",
        "refs/remotes/origin",
    ).stdout.split()
    return out[0] if out else None


def _bootstrap(dest: Path) -> None:
    if (dest / "package.json").is_file() and not (dest / "node_modules").is_dir():
        npm = shutil.which("npm")
        if not npm:
            print(f"WARN {dest.name}: package.json but no npm - build will fail")
            return
        print(f"installing npm deps for {dest.name} (one-time)")
        r = _run([npm, "install", "--ignore-scripts", "--no-audit", "--no-fund"], cwd=dest)
        if r.returncode != 0:
            print(f"WARN {dest.name}: npm install rc={r.returncode}: {r.stderr.strip()[:300]}")
    mods = dest / "node_modules"
    if mods.is_dir():
        maps = [
            f"{s}/=node_modules/{s}/\n"
            for s in ("@openzeppelin", "@uniswap", "@chainlink", "@balancer-labs",
                      "@aave", "@curvefi", "@pendle", "@layerzerolabs")
            if (mods / s).is_dir()
        ]
        if maps:
            (dest / "remappings.txt").write_text("".join(maps), encoding="utf-8")
    if not shutil.which("vyper"):
        print(f"NOTE {dest.name}: vyper not on PATH - Vyper sources will not compile")


def _update(dest: Path, branch=None) -> None:
    _git("-C", str(dest), "config", "core.sparseCheckout", "false")
    _git("-C", str(dest), "sparse-checkout", "disable")
    f = _git("-C", str(dest), "fetch", "--all", "--prune", "--tags")
    if f.returncode != 0:
        print(f"WARN {dest.name}: fetch failed: {f.stderr.strip()[:200]}")
    target = branch or _default_ref(dest)
    if not target:
        raise CloneError(f"{dest.name}: cannot resolve a default branch")
    if not target.startswith("origin/"):
        target = f"origin/{target}"
    r = _git("-C", str(dest), "reset", "--hard", target)
    if r.returncode != 0:
        print(f"WARN {dest.name}: reset to {target} failed: {r.stderr.strip()[:200]}")
        _git("-C", str(dest), "checkout", "-f", target)
    _git("-C", str(dest), "clean", "-fdx")
    if not any(dest.rglob("*.sol")):
        print(f"WARN {dest.name}: no .sol files after update - checkout incomplete")


def clone_repo(url: str, branch=None, dest=None) -> Path:
    if dest is None:
        dest, branch = branch, None
    dest = Path(dest)
    dest.parent.mkdir(parents=True, exist_ok=True)

    if dest.exists():
        if (dest / ".git").is_dir():
            _update(dest, branch)
            _bootstrap(dest)
            return dest
        if any(dest.iterdir()):
            _quarantine(dest)
        else:
            dest.rmdir()

    args = ["clone", "--filter=blob:none"]
    if branch:
        args += ["--branch", branch]
    args += [url, str(dest)]

    last = None
    for attempt in range(1, RETRIES + 1):
        if dest.exists():
            shutil.rmtree(dest, ignore_errors=True)
        r = _git(*args)
        if r.returncode == 0:
            _bootstrap(dest)
            return dest
        last = r
        time.sleep(2 * attempt)

    detail = ((last.stderr or last.stdout) if last else "").strip()[:800]
    raise CloneError(f"clone failed after {RETRIES} attempts: {url}\n{detail}")


def sparse_clone(url: str, dest, paths=None) -> Path:
    """Compat alias - sparse paths are filtered at scan time now."""
    return clone_repo(url, None, dest)

SEV = {"critical": "critical", "high": "high", "medium": "medium", "low": "low",

auto_detect_solc = true
src = "{src}"

def _norm(s):
    return SEV.get(str(s or "").lower(), "medium")


def _f(title, sev, tool, **kw):
    return {"title": title, "severity": _norm(sev), "confidence": kw.get("confidence", "medium"),
            "category": kw.get("category", "solidity"), "cwe": kw.get("cwe"),
            "file": kw.get("file"), "line": kw.get("line"), "code": kw.get("code"),
            "detail": kw.get("detail", ""), "remediation": kw.get("remediation", ""),
            "tool": tool, "economic": kw.get("economic", 2), "references": kw.get("references", []),
            "fingerprint": short_hash(f"{tool}:{kw.get('file')}:{kw.get('line')}:{title}")}


def ensure_foundry_config(repo: Path) -> bool:
    """Foundry projects ship foundry.toml. Hardhat/Truffle trees do not - bridge them
    instead of letting forge default to a src/ directory that does not exist."""
    if (repo / "foundry.toml").is_file():
        return True
    src = "contracts" if (repo / "contracts").is_dir() else "src"
    if not (repo / src).is_dir():
        return False
    (repo / "lib").mkdir(exist_ok=True)
    (repo / "foundry.toml").write_text(_TOML.format(src=src), encoding="utf-8")
    log.info("wrote foundry.toml (src=%s) for %s", src, repo.name)
    return True


def _status(res, ok_extra=False):
    if res["ok"] or ok_extra:
        return "ok"
    return f"rc={res['rc']}"


def forge_build(repo: Path, cfg):
    if not have_tool("forge"):
        return [], {"tool": "forge-build", "status": "missing"}
    if not ensure_foundry_config(repo):
        return [], {"tool": "forge-build", "status": "no sources"}
    if cfg.solidity.get("install_foundry_deps", True):
        run_cmd(["forge", "install", "foundry-rs/forge-std", "--no-commit"], cwd=repo, timeout=900)
    res = run_cmd(["forge", "build", "--sizes"], cwd=repo,
                  timeout=int(cfg.audit.get("tool_timeout", 1800)))
    oversize = "exceeds the size limit" in (res["out"] + res["err"])
    findings = []
    if not res["ok"] and not oversize:
        log.warning("forge build failed (%s):\n%s", repo.name,
                    (res["err"] or res["out"])[:2000])
        findings.append(_f("Compilation failure", "high", "forge-build", category="build",
                           detail=(res["err"] or res["out"])[:1500], economic=1))
    return findings, {"tool": "forge-build", "status": _status(res, oversize),
                      "dur": res["dur"], "err": (res["err"] or "")[:400]}

def mythril(repo: Path, cfg):

        return [], {"tool": "mythril", "status": "missing"}

    for sol in list(repo.rglob("*.sol"))[:25]:
        if any(p in {"mocks", "test", "lib", "node_modules"} for p in sol.parts):
            continue
        res = run_cmd(["myth", "analyze", str(sol), "-o", "json", "--execution-timeout", "120"],
                      cwd=repo, timeout=400)
        dur += res["dur"]
        if not res["ok"]:
            failed += 1
        try:
            for issue in json.loads(res["out"] or "{}").get("issues", []):
                findings.append(_f(issue.get("title", "mythril"), issue.get("severity"), "mythril",
                                   file=str(sol.relative_to(repo)),
                                   detail=issue.get("description", "")[:900],
                                   remediation=issue.get("remediation", "")))
        except Exception:
            continue
    status = "ok" if failed == 0 else f"partial ({failed} files failed)"
    return findings, {"tool": "mythril", "status": status, "dur": round(dur, 1)}

def _run(argv, cwd=None, timeout=900, stream=False):
    """Resolve the executable and wrap .cmd/.bat - CreateProcess cannot exec them."""
    exe = shutil.which(argv[0])
    if exe:
        argv = [exe, *argv[1:]]
        if os.name == "nt" and exe.lower().endswith((".cmd", ".bat")):
            argv = ["cmd", "/c", *argv]
    kw = dict(cwd=cwd, text=True, encoding="utf-8", errors="replace")
    if stream:
        return subprocess.run(argv, timeout=timeout, **kw)
    return subprocess.run(argv, capture_output=True, timeout=timeout, **kw)
